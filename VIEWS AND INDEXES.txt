-- 1. CREATE EMPLOYEE TABLE

CREATE TABLE Employee
(
    EmpID NUMBER PRIMARY KEY,
    EmpName VARCHAR2(50),
    Department VARCHAR2(50),
    Salary NUMBER,
    Email VARCHAR2(100)
);

-- 2. CREATE EMPLOYEE VIEW

CREATE OR REPLACE VIEW EmployeeView AS
SELECT EmpID, EmpName, Department, Salary
FROM Employee
WHERE Salary > 40000;

-- 3. CREATE STUDENT TABLE

CREATE TABLE Student
(
    StudentID NUMBER PRIMARY KEY,
    StudentName VARCHAR2(50)
);

-- 4. CREATE COURSE TABLE

CREATE TABLE Course
(
    CourseID NUMBER PRIMARY KEY,
    CourseName VARCHAR2(50)
);

-- 5. CREATE REGISTRATION TABLE

CREATE TABLE Registration
(
    RegistrationID NUMBER PRIMARY KEY,
    StudentID NUMBER,
    CourseID NUMBER,
    Grade VARCHAR2(5)
);

-- 6. CREATE STUDENT COURSE VIEW

CREATE OR REPLACE VIEW StudentCourseView AS
SELECT s.StudentName, c.CourseName, r.Grade
FROM Student s
INNER JOIN Registration r
ON s.StudentID = r.StudentID
INNER JOIN Course c
ON r.CourseID = c.CourseID;

PROMPT
PROMPT 2. STUDENT COURSE VIEW

SELECT * FROM StudentCourseView
ORDER BY StudentName;
