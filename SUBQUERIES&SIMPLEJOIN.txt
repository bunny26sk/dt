
-- Create Publisher Table
CREATE TABLE Publisher (
    Publisher_ID NUMBER(5) PRIMARY KEY,
    Publisher_Name VARCHAR2(30)
);

-- Create LibraryBook Table
CREATE TABLE LibraryBook (
    Book_ID NUMBER(5) PRIMARY KEY,
    Book_Name VARCHAR2(30),
    Price NUMBER(10),
    Publisher_ID NUMBER(5)
);

-- Display Publisher Table
SELECT * FROM Publisher;

-- Display LibraryBook Table
SELECT * FROM LibraryBook;

-- 1. WHERE Clause
SELECT *
FROM LibraryBook
WHERE Price > 400;

-- 2. Subquery Using IN
SELECT Book_Name, Price
FROM LibraryBook
WHERE Publisher_ID IN (
    SELECT Publisher_ID
    FROM Publisher
    WHERE Publisher_Name IN
    ('Sun Publications', 'Tech Publications')
);

-- 3. INNER JOIN
SELECT LibraryBook.Book_Name,
       Publisher.Publisher_Name
FROM LibraryBook
INNER JOIN Publisher
ON LibraryBook.Publisher_ID = Publisher.Publisher_ID;

-- 4. LEFT JOIN
SELECT LibraryBook.Book_Name,
       Publisher.Publisher_Name
FROM LibraryBook
LEFT JOIN Publisher
ON LibraryBook.Publisher_ID = Publisher.Publisher_ID;
