To create a Product table and perform DDL and DML operations using SQL commands.

    Product_ID NUMBER(5) PRIMARY KEY,

SELECT * FROM Product;

-- Delete a Product
DELETE FROM Product
WHERE Product_ID = 103;

-- Display Records After DELETE
SELECT * FROM Product;

-- Add a New Column
ALTER TABLE Product
ADD Stock NUMBER(5);

-- Display Table Structure
DESC Product;

-- Update Stock Values
UPDATE Product
SET Stock = 20
WHERE Product_ID = 101;

UPDATE Product
SET Stock = 35
WHERE Product_ID = 102;

-- Display Records After ALTER TABLE
SELECT * FROM Product;

-- Remove All Records
TRUNCATE TABLE Product;

-- Display Empty Table
SELECT * FROM Product;

EXIT;

Aim

To execute SQL queries using the WHERE clause, subquery with IN, INNER JOIN and LEFT JOIN using Library and Publisher details.

Create a Publisher table and a LibraryBook table.

Use a subquery with IN to retrieve books from selected publishers.

Perform a LEFT JOIN to display all books, including books without a matching publisher.
Display the results.
Stop the program.

-- Create Publisher Table

    Publisher_ID NUMBER(5) PRIMARY KEY,
    Publisher_Name VARCHAR2(30)
);

-- Create LibraryBook Table

    Book_ID NUMBER(5) PRIMARY KEY,

SELECT * FROM LibraryBook;

-- 1. WHERE Clause
SELECT *
FROM LibraryBook

-- 2. Subquery Using IN

FROM LibraryBook
WHERE Publisher_ID IN (
    SELECT Publisher_ID
    FROM Publisher
    WHERE Publisher_Name IN
    ('Sun Publications', 'Tech Publications')
);

-- 3. INNER JOIN

       Publisher.Publisher_Name

INNER JOIN Publisher
ON LibraryBook.Publisher_ID = Publisher.Publisher_ID;

       Publisher.Publisher_Name
FROM LibraryBook
LEFT JOIN Publisher
ON LibraryBook.Publisher_ID = Publisher.Publisher_ID;

PROGRAM 6 – USER DEFINED FUNCTIONS AND STORED PROCEDURES

Aim:

To create and execute user-defined functions and stored procedures in SQL to calculate employee bonuses, retrieve department details, and insert employee records.

Procedure:
1. Create the Employee table.
2. Insert employee records into the table.
3. Create a scalar function to calculate employee bonus.
4. Execute the function and display employee bonus details.
5. Create a REF CURSOR function to retrieve department-wise employee details.
6. Create a stored procedure to insert employee records.
7. Execute the procedure and display the results.

Source Code: 

SET SERVEROUTPUT ON;

-- Create Employee Table

CREATE TABLE Employee
(
    EmpID NUMBER PRIMARY KEY,
    EmpName VARCHAR2(50),
    Department VARCHAR2(50),
    Salary NUMBER(10,2),
    Email VARCHAR2(100)
);

-- Insert Employee Record

INSERT INTO Employee VALUES
(101, 'Rajesh Kumar', 'IT', 50000,
 'rajesh@company.com');

COMMIT;


-- 1. Scalar Function

CREATE OR REPLACE FUNCTION CalculateBonus
(
    p_Salary NUMBER
)
RETURN NUMBER
IS
BEGIN
    RETURN p_Salary * 0.10;
END;
/

-- Execute Scalar Function

SELECT EmpName, Salary,
CalculateBonus(Salary) AS Bonus
FROM Employee;


-- 2. REF CURSOR Function

CREATE OR REPLACE FUNCTION GetEmployeesByDept
(
    p_Dept VARCHAR2
)
RETURN SYS_REFCURSOR
IS
    emp_cursor SYS_REFCURSOR;
BEGIN
    OPEN emp_cursor FOR
    SELECT EmpID, EmpName, Salary
    FROM Employee
    WHERE Department = p_Dept;

    RETURN emp_cursor;
END;
/

-- Execute REF CURSOR Function

VARIABLE result REFCURSOR;

EXEC :result := GetEmployeesByDept('IT');

PRINT result;


-- 3. Stored Procedure

CREATE OR REPLACE PROCEDURE AddEmployee
(
    p_EmpID NUMBER,
    p_EmpName VARCHAR2,
    p_Department VARCHAR2,
    p_Salary NUMBER,
    p_Email VARCHAR2
)
IS
BEGIN
    IF p_Salary > 0 THEN

        INSERT INTO Employee
        (
            EmpID,
            EmpName,
            Department,
            Salary,
            Email
        )
        VALUES
        (
            p_EmpID,
            p_EmpName,
            p_Department,
            p_Salary,
            p_Email
        );

        DBMS_OUTPUT.PUT_LINE(
            'Employee added successfully'
        );

    ELSE

        DBMS_OUTPUT.PUT_LINE(
            'Invalid Salary'
        );

    END IF;

END;
/

-- Execute Stored Procedure

EXEC AddEmployee(106, 'Vikram Desai', 'IT', 45000, 'vikram@company.com');

COMMIT;

Output

2. REF CURSOR Function

101      Rajesh Kumar     50000


3. Stored Procedure
Employee added successfully

PL/SQL procedure successfully completed.

Thus, user-defined functions and stored procedures were successfully created and executed to calculate employee bonuses, retrieve department details, and insert employee records.

PROGRAM 7: TRANSACTIONS AND DCL AND TCL COMMANDS

Aim

To execute database transactions using COMMIT, ROLLBACK, SAVEPOINT and implement DCL commands to maintain database security and consistency.

Procedure

1. Select the Employee database table.
2. Begin a transaction using BEGIN TRANSACTION.
3. Update employee salaries and commit the changes.
4. Perform a DELETE operation and use ROLLBACK.
5. Create a savepoint and perform additional operations.
6. Execute GRANT and REVOKE commands.
7. Display the final records and stop the program.

Source Code

SET SERVEROUTPUT ON;

-- 1. COMMIT

PROMPT COMMIT:

UPDATE Employee
SET Salary = Salary + 5000
WHERE Department = 'IT';

COMMIT;

BEGIN
    DBMS_OUTPUT.PUT_LINE('Transaction committed successfully.');
END;
/

-- 2. ROLLBACK

PROMPT ROLLBACK:

DELETE FROM Employee
WHERE EmpID = 999;

ROLLBACK;

BEGIN
    DBMS_OUTPUT.PUT_LINE('Transaction rolled back successfully.');
END;
/

-- 3. SAVEPOINT

PROMPT SAVEPOINT:

UPDATE Employee
SET Salary = Salary + 2000
WHERE EmpID = 101;

SAVEPOINT sp1;

UPDATE Employee
SET Salary = Salary + 3000
WHERE EmpID = 101;

ROLLBACK TO sp1;

COMMIT;

BEGIN
    DBMS_OUTPUT.PUT_LINE('Transaction rolled back to savepoint sp1.');
END;
/

-- 4. DCL COMMANDS

PROMPT DCL:

GRANT SELECT ON Employee TO PUBLIC;

BEGIN
    DBMS_OUTPUT.PUT_LINE('GRANT command executed successfully.');
END;
/

REVOKE SELECT ON Employee FROM PUBLIC;

BEGIN
    DBMS_OUTPUT.PUT_LINE('REVOKE command executed successfully.');
END;
/

-- DISPLAY RECORDS

SELECT * FROM Employee;

COMMIT;

PROGRAM 8: SQL TRIGGERS

Aim

To create and execute SQL triggers for INSERT, UPDATE and DELETE operations to maintain data integrity and enforce business rules.

Procedure

1. Select the database containing the Employee table.
2. Create an AuditLog table.
3. Create an AFTER INSERT trigger to record employee additions.
4. Create an AFTER UPDATE trigger to record salary changes.
5. Create an AFTER DELETE trigger to record deleted employees.
6. Execute INSERT, UPDATE and DELETE operations.
7. Verify the trigger results using the AuditLog table

Source Code

-- PROGRAM 8: SQL TRIGGERS
-- Aim: To create and execute SQL triggers for INSERT,
-- UPDATE and DELETE operations.

-- STEP 1: CONNECT AS SYSDBA

CONN / AS SYSDBA

-- STEP 2: CREATE STUDENT USER

CREATE USER STUDENT IDENTIFIED BY STUDENT;

GRANT CONNECT, RESOURCE TO STUDENT;

ALTER USER STUDENT QUOTA UNLIMITED ON USERS;

-- STEP 3: CONNECT AS STUDENT

CONN STUDENT/STUDENT

-- STEP 4: CREATE EMPLOYEE TABLE

CREATE TABLE Employee
(
    EmpID NUMBER PRIMARY KEY,
    EmpName VARCHAR2(50),
    Department VARCHAR2(50),
    Salary NUMBER,
    Email VARCHAR2(100)
);

-- STEP 5: INSERT SAMPLE RECORDS

INSERT INTO Employee
VALUES (101, 'Arun', 'IT', 40000, 'arun@gmail.com');

INSERT INTO Employee
VALUES (102, 'Priya', 'HR', 35000, 'priya@gmail.com');

COMMIT;

-- STEP 6: CREATE AUDITLOG TABLE

CREATE TABLE AuditLog
(
    ActionType VARCHAR2(20),
    EmployeeID NUMBER,
    EmployeeName VARCHAR2(50)
);

-- STEP 7: CREATE INSERT TRIGGER

CREATE OR REPLACE TRIGGER trg_Insert
AFTER INSERT ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO AuditLog
    VALUES ('INSERT', :NEW.EmpID, :NEW.EmpName);
END;
/

-- STEP 8: CREATE UPDATE TRIGGER

CREATE OR REPLACE TRIGGER trg_Update
AFTER UPDATE ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO AuditLog
    VALUES ('UPDATE', :NEW.EmpID, :NEW.EmpName);
END;
/

-- STEP 9: CREATE DELETE TRIGGER

CREATE OR REPLACE TRIGGER trg_Delete
AFTER DELETE ON Employee
FOR EACH ROW
BEGIN
    INSERT INTO AuditLog
    VALUES ('DELETE', :OLD.EmpID, :OLD.EmpName);
END;
/

-- STEP 10: TEST INSERT OPERATION

INSERT INTO Employee
(EmpID, EmpName, Department, Salary, Email)
VALUES
(107, 'Neha Sharma', 'IT', 48000,
'neha@company.com');

-- STEP 11: TEST UPDATE OPERATION

UPDATE Employee
SET Salary = 52000
WHERE EmpID = 107;

-- STEP 12: TEST DELETE OPERATION

DELETE FROM Employee
WHERE EmpID = 107;

-- STEP 13: SAVE CHANGES

COMMIT;

-- STEP 14: DISPLAY AUDIT LOG

SELECT * FROM AuditLog;

-- STEP 15: CHECK TRIGGER STATUS

SELECT TRIGGER_NAME, STATUS
FROM USER_TRIGGERS;

-- END OF PROGRAM

Output

Result

Thus, SQL triggers were successfully created and executed for INSERT, UPDATE and DELETE operations to maintain database integrity and record employee activities.
