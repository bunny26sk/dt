
PROGRAM 1: DDL AND DML OPERATIONS

Aim

To create a Product table and perform DDL and DML operations using SQL commands.

Procedure

Create a database named ShopDB.
Create a Product table with suitable fields.
Insert product records into the table.
Display the records using SELECT.
Update the price of a product using UPDATE.
Delete a product using DELETE.
Add a new column using ALTER TABLE.
Display the table structure using DESC.
Remove all records using TRUNCATE.
Stop the program.


Source Code

-- Create Database Table
-- Oracle uses the connected user schema.
-- Create Product Table

CREATE TABLE Product (
    Product_ID NUMBER(5) PRIMARY KEY,
    Product_Name VARCHAR2(30),
    Category VARCHAR2(20),
    Price NUMBER(10)
);

-- Insert Product Records
INSERT INTO Product VALUES (101, 'Keyboard', 'Electronics', 800);

INSERT INTO Product VALUES (102, 'Mouse', 'Electronics', 500);

INSERT INTO Product VALUES (103, 'Notebook', 'Stationery', 100);

-- Display Initial Records
SELECT * FROM Product;

-- Update Product Price
UPDATE Product
SET Price = 900
WHERE Product_ID = 101;

-- Display Records After UPDATE
SELECT * FROM Product;

-- Delete a Product
DELETE FROM Product
WHERE Product_ID = 103;

-- Display Records After DELETE
SELECT * FROM Product;

-- Add a New Column
ALTER TABLE Product
ADD Stock NUMBER(5);

-- Display Table Structure
DESC Product;

-- Update Stock Values
UPDATE Product
SET Stock = 20
WHERE Product_ID = 101;

UPDATE Product
SET Stock = 35
WHERE Product_ID = 102;

-- Display Records After ALTER TABLE
SELECT * FROM Product;

-- Remove All Records
TRUNCATE TABLE Product;

-- Display Empty Table
SELECT * FROM Product;

-- Stop the Program
EXIT;

After DELETE

PROGRAM 2: FOREIGN KEY AND REFERENTIAL INTEGRITY


Aim

To create related Customer and Orders tables and implement foreign key constraints to maintain referential integrity.

Procedure

Create a database named SalesDB.
Create a Customer table with Customer_ID as the primary key.
Create an Orders table with Order_ID as the primary key and Customer_ID as a foreign key.
Insert customer records.
Insert orders for existing customers.
Display the records from both tables.
Attempt to insert an order for a customer who does not exist.
Verify the foreign key constraint.
Stop the program.

Source Code

-- Create Customer Table
CREATE TABLE Customer (
    Customer_ID NUMBER(5) PRIMARY KEY,
    Customer_Name VARCHAR2(30) NOT NULL,
    City VARCHAR2(30)
);

-- Create Orders Table
CREATE TABLE Orders (
    Order_ID NUMBER(5) PRIMARY KEY,
    Order_Date DATE,
    Amount NUMBER(10),
    Customer_ID NUMBER(5),
    CONSTRAINT fk_customer
    FOREIGN KEY (Customer_ID)
    REFERENCES Customer(Customer_ID)
);


-- Insert Customer Records
INSERT INTO Customer
VALUES (1, 'Arun', 'Chennai');

INSERT INTO Customer
VALUES (2, 'Priya', 'Madurai');

INSERT INTO Customer
VALUES (3, 'Kumar', 'Coimbatore');

VALUES (103, TO_DATE('15-01-2026','DD-MM-YYYY'), 3200, 1);

INSERT INTO Orders
VALUES (104, TO_DATE('20-01-2026','DD-MM-YYYY'), 1500, 3);

-- Display Customer Table
SELECT * FROM Customer;

-- Display Orders Table

-- Test Referential Integrity

PROGRAM 3: WHERE CLAUSE AND AGGREGATE FUNCTIONS

Aim

To retrieve book details using the WHERE clause and calculate aggregate values using COUNT(), SUM(), AVG(), MAX() and MIN().

Procedure

Create a Book table.
Insert sample book records.
Retrieve books with a price greater than 300 using WHERE.
Calculate the total number of books.
Calculate the total and average price.
Find the highest and lowest prices using aggregate functions.
Display the results.
Stop the program.

Source Code

CREATE TABLE Book (
Book_ID NUMBER(5) PRIMARY KEY,
Book_Name VARCHAR2(40),
Category VARCHAR2(20),
Price NUMBER(10)
);
-- Insert Book Records
INSERT INTO Book VALUES (1, 'SQL Basics', 'Database', 450);
INSERT INTO Book VALUES (2, 'Python Guide', 'Programming', 600);
INSERT INTO Book VALUES (3, 'Java Basics', 'Programming', 500);
INSERT INTO Book VALUES (4, 'Web Design', 'Web', 300); 
INSERT INTO Book VALUES (5, 'DBMS Guide', 'Database', 400);
-- Display All Book Records
SELECT * FROM Book;
-- Display Books Priced Above 300
SELECT *
FROM Book
WHERE Price > 300;
-- Aggregate Functions for Database Category
SELECT
COUNT(*) AS Total_Books,
SUM(Price) AS Total_Price,
AVG(Price) AS Average_Price,
MAX(Price) AS Highest_Price,
MIN(Price) AS Lowest_Price
FROM Book
WHERE Category = 'Database';


-- Stop the Program
EXIT;

PROGRAM 4: SUBQUERIES AND SIMPLE JOINS

Aim

To execute SQL queries using the WHERE clause, subquery with IN, INNER JOIN and LEFT JOIN using Library and Publisher details.

Procedure

Create a Publisher table and a LibraryBook table.
Insert records into both tables.
Retrieve books with a price greater than 400.
Use a subquery with IN to retrieve books from selected publishers.
Perform an INNER JOIN to display books with matching publisher details.
Perform a LEFT JOIN to display all books, including books without a matching publisher.
Display the results.
Stop the program.

Source Code

-- Create Publisher Table
CREATE TABLE Publisher (
    Publisher_ID NUMBER(5) PRIMARY KEY,
    Publisher_Name VARCHAR2(30)
);

-- Create LibraryBook Table
CREATE TABLE LibraryBook (
    Book_ID NUMBER(5) PRIMARY KEY,
    Book_Name VARCHAR2(30),
    Price NUMBER(10),
    Publisher_ID NUMBER(5)
);

-- Insert Publisher Records
INSERT INTO Publisher VALUES (10, 'Sun Publications');

INSERT INTO Publisher VALUES (20, 'Tech Publications');

INSERT INTO Publisher VALUES (30, 'Global Books');

-- Insert LibraryBook Records
INSERT INTO LibraryBook VALUES (1, 'SQL Guide', 450, 10);

INSERT INTO LibraryBook VALUES (2, 'Python Guide', 600, 20);

INSERT INTO LibraryBook VALUES (3, 'Java Guide', 500, 10);

INSERT INTO LibraryBook VALUES (4, 'Web Design', 350, 30);

INSERT INTO LibraryBook VALUES (5, 'C Programming', 400, NULL);

-- Display Publisher Table
SELECT * FROM Publisher;

-- Display LibraryBook Table
SELECT * FROM LibraryBook;

-- 1. WHERE Clause
SELECT *
FROM LibraryBook
WHERE Price > 400;

-- 2. Subquery Using IN
SELECT Book_Name, Price
FROM LibraryBook
WHERE Publisher_ID IN (
    SELECT Publisher_ID
    FROM Publisher
    WHERE Publisher_Name IN
    ('Sun Publications', 'Tech Publications')
);

-- 3. INNER JOIN
SELECT LibraryBook.Book_Name,
       Publisher.Publisher_Name
FROM LibraryBook
INNER JOIN Publisher
ON LibraryBook.Publisher_ID = Publisher.Publisher_ID;

-- 4. LEFT JOIN
SELECT LibraryBook.Book_Name,
       Publisher.Publisher_Name
FROM LibraryBook
LEFT JOIN Publisher
ON LibraryBook.Publisher_ID = Publisher.Publisher_ID;

-- Stop the Program
EXIT;

PROGRAM 5: NATURAL JOIN, EQUI JOIN AND OUTER JOINS

Aim

To perform Natural Join, Equi Join, Left Outer Join and Right Outer Join using Vehicle and VehicleType tables.

Procedure

Create Vehicle and VehicleType tables.
Insert sample vehicle and vehicle type records.
Perform a NATURAL JOIN using the common Type_ID column.
Perform an EQUI JOIN using the equality operator.
Perform a LEFT OUTER JOIN to display all vehicles and their types.
Perform a RIGHT OUTER JOIN to display all vehicle types and matching vehicles.
Display the results.
Stop the program.

Source Code

-- Create Vehicle Table
CREATE TABLE Vehicle (
    Vehicle_ID NUMBER(5) PRIMARY KEY,
    Vehicle_Name VARCHAR2(30),
    Type_ID NUMBER(5)
);

-- Create VehicleType Table
CREATE TABLE VehicleType (
    Type_ID NUMBER(5),
    Type_Name VARCHAR2(30)
);

-- Insert Vehicle Records
INSERT INTO Vehicle VALUES (1, 'Honda City', 10);

INSERT INTO Vehicle VALUES (2, 'Royal Enfield', 20);

INSERT INTO Vehicle VALUES (3, 'Hyundai i20', 10);

INSERT INTO Vehicle VALUES (4, 'Hero Cycle', 30);


-- Insert VehicleType Records
INSERT INTO VehicleType VALUES (10, 'Car');

INSERT INTO VehicleType VALUES (20, 'Bike');

INSERT INTO VehicleType VALUES (40, 'Bus');

-- Display Vehicle Table
SELECT * FROM Vehicle;

-- Display VehicleType Table
SELECT * FROM VehicleType;

-- 1. NATURAL JOIN
SELECT *
FROM Vehicle
NATURAL JOIN VehicleType;

-- 2. EQUI JOIN
SELECT Vehicle.Vehicle_ID,
       Vehicle.Vehicle_Name,
       Vehicle.Type_ID,
       VehicleType.Type_Name
FROM Vehicle
JOIN VehicleType
ON Vehicle.Type_ID = VehicleType.Type_ID;

-- 3. LEFT OUTER JOIN
SELECT Vehicle.Vehicle_ID,
       Vehicle.Vehicle_Name,
       Vehicle.Type_ID,
       VehicleType.Type_Name
FROM Vehicle
LEFT OUTER JOIN VehicleType
ON Vehicle.Type_ID = VehicleType.Type_ID;

-- 4. RIGHT OUTER JOIN
SELECT Vehicle.Vehicle_ID,
       Vehicle.Vehicle_Name,
       VehicleType.Type_ID,
       VehicleType.Type_Name
FROM Vehicle
RIGHT OUTER JOIN VehicleType
ON Vehicle.Type_ID = VehicleType.Type_ID;

-- Stop the Program
EXIT;
